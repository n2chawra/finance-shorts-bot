def E(*words):
    ev = {}
    for w in words:
        name = re.sub(r"[^a-z0-9]", "", w.lower()) or "w"
        while name in ev:
            name += "x"
        ev[name] = {"word": w}
    return ev


def nm(w):
    return re.sub(r"[^a-z0-9]", "", w.lower())


def A(at, pose, expr, **kw):
    d = {"sec": 0} if at == 0 else {"at": nm(at)}
    d.update({"pose": pose, "expression": expr}, **kw)
    return d


def C(at, size, on, move=True):
    d = {"sec": 0, "size": size, "on": on} if at == 0 else {"at": nm(at), "size": size, "on": on}
    if at != 0 and move:
        d["move"] = True
    return d


def S(kind, at, **kw):
    d = {"kind": kind, "at": at if at in ("start", "end", "speech") else nm(at)}
    d.update(kw)
    return d


def hud(level=None, counter=None):
    h = {}
    if level is not None:
        h["level"] = level
    if counter:
        h["counter"] = counter
    else:
        h["noCounter"] = True
    return h


def ctr(name, key, label, at):
    return {"value": f"@{name}.{key}", "text": f"{{{name}_{key}}}", "label": label, "at": nm(at)}


BEATS = []


def beat(id, scene, speaker, say, ev, sfx, params, role="level", tr="cut"):
    b = {"id": id, "scene": scene}
    if role:
        b["role"] = role
    b.update({"speaker": speaker, "transition": tr, "say": say, "events": ev, "sfx": sfx, "params": params})
    BEATS.append(b)


def shot(id, who, say, ev, set, acting, cam, level, word=None, objects=None, tr="cut", spot="center", extra_actors=None, counter=None, role="level", sfx=None):
    p = {"set": set, "actors": [{"who": who, "acting": acting, "spot": spot}] + (extra_actors or [])}
    if objects:
        p["objects"] = objects
    p["camera"] = cam
    if word:
        p["word"] = {"text": word[0], "at": nm(word[1]), "color": word[2], "where": "top"}
    p["hud"] = hud(level, counter)
    beat(id, "shot", who, say, ev, sfx or [], p, role=role, tr=tr)


def build(id, say, ev, world, items, actions, cam, level, actors=None, speaker="narrator", tr="match", counter=None, role="level", sfx=None):
    p = {"world": world}
    if actors:
        p["actors"] = actors
        who = actors[0]["who"]
        items = [{k: v for k, v in i.items() if k != "label"} for i in items]   # labels run off-frame on a close-up of the cast
        cam = [k if k.get("on") in ("room", "all") else dict(k, on=who) if k.get("size") in ("close", "extreme") else dict(k, size="medium", on="all") for k in cam]
    p.update({"items": items, "actions": actions, "camera": cam, "hud": hud(level, counter)})
    beat(id, "build", speaker, say, ev, sfx or [], p, role=role, tr=tr)


def I(id, prop, x, size, hidden=False, **kw):
    d = {"id": id, "prop": prop, "x": x, "size": size}
    if hidden:
        d["hidden"] = True
    d.update(kw)
    return d


def act(at, do, target, **kw):
    d = {"at": nm(at), "do": do, "target": target}
    d.update(kw)
    return d


def actor(who, spot, acting, **kw):
    d = {"who": who, "acting": acting, "spot": spot}
    d.update(kw)
    return d


def docu(id, scene, say, ev, params, level, tr="whip", role="level", sfx=None, speaker="narrator"):
    params = dict(params)
    params["hud"] = hud(level, params.pop("counter", None))
    beat(id, scene, speaker, say, ev, sfx or [], params, role=role, tr=tr)


def chapter(n, kicker, title, say, ev):
    beat(f"ch{n}", "chapter", "narrator", say, ev, [S("whoosh", "start")], {"kicker": kicker, "title": title, "hud": hud(n)}, role=None, tr="fade")
